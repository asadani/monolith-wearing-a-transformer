/* Flow on screen, paginated in print. The page-break rules above are untouched,
   so the PDF still starts every chapter on a fresh page; what changes here is
   only that the reader gets one continuous column instead of a stack of cards.

   The column is applied to every top-level section, not just .sheet, because
   book-forge's flow matter arrives as a sibling <section class="bf-matter">
   and its stylesheet deliberately declares no width -- it expects the host to
   lay it out. Without this the copyright and about pages run full-bleed. */

  body{
    background:var(--paper);
    padding:0 0 72px;
  }

  body > section{
    max-width:760px;

    padding:0 32px;

    box-shadow:none;
  }

     measure, so let the text fill it. */

  .titlepage .sub,
  .titlepage .desc,
  .titlepage .verdict,
  .titlepage .byline,
  .chapter h2,
